\section{Problema}
% ============================================================

\begin{frame}{O PSP e sua formulação}
    \footnotesize
    % \setlength{\abovedisplayskip}{4pt}\setlength{\belowdisplayskip}{4pt}

    \begin{align*}
        \max \quad & \frac{\sum_{o \in \mathcal{O}} \sum_{i \in \mathcal{I}_o} q_{oi}\, x_o}{\sum_{a \in \mathcal{A}} y_a}
                   &                                                                                                                             & \text{(unidades por corredor)}    \\[2pt]
        \text{s.a.} \quad
                   & \mathrm{LB} \leq \textstyle\sum_{o \in \mathcal{O}} \sum_{i \in \mathcal{I}_o} q_{oi}\, x_o \leq \mathrm{UB}
                   &                                                                                                                             & \text{(tamanho da \textit{wave})} \\
                   & \textstyle\sum_{o \in \mathcal{O}} q_{oi}\, x_o \leq \sum_{a \in \mathcal{A}} e_{ai}\, y_a, \quad \forall i \in \mathcal{I}
                   &                                                                                                                             & \text{(disponibilidade)}          \\
                   & x_o \in \{0,1\},\ \forall o \in \mathcal{O}; \quad y_a \in \{0,1\},\ \forall a \in \mathcal{A}
                   &                                                                                                                             & \text{(integralidade)}
    \end{align*}

    \medskip
    \begin{itemize}
        \item $\mathcal{O}$: pedidos; \ $\mathcal{I}_o$: itens do pedido $o$; \ $\mathcal{I}$: itens; \ $\mathcal{A}$: corredores
        \item $q_{oi}$: unidades do item $i$ pedidas por $o$; \ $e_{ai}$: unidades do item $i$ no corredor $a$
        \item $x_o = 1$ se o pedido $o$ compõe a \textit{wave}; \ $y_a = 1$ se o corredor $a$ é visitado
    \end{itemize}

\end{frame}

        \caption{Exemplo do PSP: \textit{backlog}, corredores e três \textit{waves} viáveis ($\mathrm{LB} = 5$, $\mathrm{UB} = 12$).}
    \end{figure}
\end{frame}

\section{Métodos}
% ============================================================

\begin{frame}{Objetivos}
    \footnotesize
    \textbf{Objetivo Geral:} Implementar e avaliar experimentalmente abordagens heurísticas e
    meta-heurísticas para o Problema de Seleção de Pedidos Ótima (PSP) com as
    instâncias reais do Desafio SBPO 2025 como \textit{benchmark} padronizado.

    \medskip
    \textbf{Objetivos Específicos:}
    \begin{enumerate}
        \item Formalizar o PSP e situá-lo na literatura do \textit{Order Batching Problem}
        \item Desenvolver a heurística construtiva \textit{OrderBased}, com composição definida pela fronteira de Pareto
        \item Desenvolver a heurística construtiva \textit{AisleBased} e compará-la ao \textit{OrderBased}
        \item Propor e implementar a meta-heurística ILS-VND, partindo da solução do \textit{AisleBased}
        \item Avaliar os métodos nas instâncias A, B e X frente às MSC e ao método de referência
    \end{enumerate}
\end{frame}

\begin{frame}{\textit{OrderBased}}
    \small
    \textbf{\textit{OrderBased}:} heurística construtiva que escolhe primeiro os pedidos da \textit{wave} e só depois
    os corredores que cobrem a sua demanda.

    \medskip
    \begin{figure}
        \centering
        \includegraphics[width=\textwidth,height=0.58\textheight,keepaspectratio]{Fig/fluxo_orderbased.pdf}
        \caption{Fluxograma da heurística \textit{OrderBased}.}
    \end{figure}
\end{frame}

\begin{frame}{\textit{AisleBased}}
    \small
    \textbf{\textit{AisleBased}:} escolhe primeiro os corredores e depois empacota os pedidos que o estoque acumulado atende.

    \medskip
    \begin{figure}
        \centering
        \includegraphics[width=\textwidth,height=0.58\textheight,keepaspectratio]{Fig/fluxo_aislebased.pdf}
        \caption{Fluxograma da heurística \textit{AisleBased}.}
    \end{figure}
\end{frame}

\begin{frame}{ILS-VND}
    \small
    \textbf{ILS-VND:} meta-heurística que refina a solução do \textit{AisleBased}, alternando perturbação guiada
    e busca local VND.

    \medskip
    \begin{figure}
        \centering
        \includegraphics[width=\textwidth,height=0.58\textheight,keepaspectratio]{Fig/fluxo_ilsvnd.pdf}
        \caption{Fluxograma da meta-heurística ILS-VND.}
    \end{figure}
\end{frame}

\begin{frame}{Protocolo experimental}
    % \footnotesize

    \textbf{RPD} (desvio percentual relativo à MSC)
    \[
        \text{RPD} = \frac{f^\ast - f}{f^\ast} \times 100\%
    \]
    $f^\ast$: Melhor Solução Conhecida (MSC);

    $f$: objetivo obtido pelo algoritmo.

    \bigskip

    \footnotesize
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \textbf{Friedman} (teste global)
            \[
                \chi^2_F = \frac{12N}{k(k+1)} \Bigl[\sum_{j=1}^{k} R_j^2 - \frac{k(k+1)^2}{4}\Bigr]
            \]
            Ranqueia os $k = 3$ algoritmos em cada uma das $N = 50$ instâncias;

            $R_j$: \textit{ranking} médio.

            $H_0$: algoritmos equivalentes.
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Wilcoxon} (par a par, sobre o RPD)
            \[
                W = \min\Bigl(\sum_{d_i > 0} \text{posto}(|d_i|),\ \sum_{d_i < 0} \text{posto}(|d_i|)\Bigr)
            \]
            $d_i$: diferença de RPD do par na instância $i$. Usa sinal e magnitude; $p$-valores ajustados por Holm ($\alpha = 0{,}05$).
        \end{column}
    \end{columns}
\end{frame}
